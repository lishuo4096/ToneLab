cmake_minimum_required(VERSION 3.20)
project(ToneLab VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "ToneLab requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
add_library(tone_core STATIC src/dsp.cpp src/device.cpp src/audio.cpp third_party/rtaudio/RtAudio.cpp
 third_party/rtaudio/include/asio.cpp third_party/rtaudio/include/asiodrivers.cpp
 third_party/rtaudio/include/asiolist.cpp third_party/rtaudio/include/iasiothiscallresolver.cpp)
target_include_directories(tone_core PUBLIC include PRIVATE third_party/rtaudio third_party/rtaudio/include)
target_compile_definitions(tone_core PRIVATE __WINDOWS_DS__ __WINDOWS_ASIO__ NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(tone_core PUBLIC dsound winmm ole32 uuid propsys ksuser avrt)
add_executable(tonelab-cli src/cli.cpp)
target_link_libraries(tonelab-cli PRIVATE tone_core)
add_executable(ToneLab WIN32 src/gui_main.cpp src/ui.cpp
 third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp
 third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp
 third_party/imgui/backends/imgui_impl_win32.cpp third_party/imgui/backends/imgui_impl_dx11.cpp)
target_include_directories(ToneLab PRIVATE third_party/imgui third_party/imgui/backends)
target_link_libraries(ToneLab PRIVATE tone_core d3d11 dxgi d3dcompiler dwmapi imm32 windowscodecs)
if(MINGW)
  target_link_options(ToneLab PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_options(tonelab-cli PRIVATE -static -static-libgcc -static-libstdc++)
endif()
enable_testing()
add_executable(dsp-tests tests/dsp_tests.cpp src/dsp.cpp)
target_include_directories(dsp-tests PRIVATE include)
add_test(NAME dsp COMMAND dsp-tests)
add_executable(loopback-test tests/loopback.cpp)
target_link_libraries(loopback-test PRIVATE tone_core)
foreach(app ToneLab tonelab-cli dsp-tests loopback-test)
  target_sources(${app} PRIVATE resources/tonelab.rc)
  target_include_directories(${app} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/assets)
endforeach()
